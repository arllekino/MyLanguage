class Robot {
    let name: String
    let id: Int

    init(name: String, id: Int) {
        self.name = name
        self.id = id
    }

    func identify() {
        print("I am robot")
        print(self.name)
        print(self.id)
    }
}

let r2d2 = Robot("R2-D2", 101)
r2d2.identify()
