class BankAccount {
    private let balance: Double

    private let tax: Double {
        return self.balance * 0.13
    }

    init(balance: Double) {
        self.balance = balance
    }

    internal func deposit(amount: Double): Void {
        self.balance = self.balance + amount
        print("Tax will be:")
        print(self.tax)
    }
}

let account = BankAccount(1000.0)
account.deposit(500.0) // 195
